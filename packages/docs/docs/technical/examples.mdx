# Examples

## Confirmation

<ConfirmationDialog />

```jsx
const handleCompletion = (result) => {
  setOpen(false);
  if (result === "success") {
    setTimeout(() => {
      setConfirmed(true);
    }, 1000);
    setConfirmed(false);
  } else if (result === "failure") {
    setTimeout(() => {
      setRejected(true);
    }, 1000);
  }
  setTimeout(() => {
    setRejected(false);
    setConfirmed(false);
  }, 2000);
};
return (
  <Dialog
    open={open}
    onClose={() => setOpen(false)}
    title="Confirm Deletion"
    role="alertdialog"
    onComplete={handleCompletion}
  >
    <DialogBody>
      Are you sure you want to permanently delete this item? This action cannot
      be undone.
    </DialogBody>
    <DialogActions>
      <DialogCloseButton>Cancel</DialogCloseButton>
      <DialogSuccessButton>Delete Permanently</DialogSuccessButton>
    </DialogActions>
  </Dialog>
);
```
